| 死连接清理与历史堆积 | 半开连接在心跳周期内被剔除；历史消息扩到 10 倍后进程驻留量仍可控 | 制造 TCP 半开场景看应用层心跳能否主动发现，同时长跑一个房间统计落盘量 | 剔除不及时就缩短心跳周期并设超时次数，存储侧撑不住（文件涨、连接排队或写锁等待）就按房间建索引并清理过期历史 |

import { WebSocketServer } from 'ws';

const wss = new WebSocketServer({ port: 8080 });
wss.on('connection', (socket) => {
  socket.on('message', (raw) => {
    const msg = JSON.parse(raw.toString());
    const frame = JSON.stringify({ id: msg.id, type: 'message', text: msg.text });
    for (const client of wss.clients) {
      if (client.readyState === client.OPEN) client.send(frame);
    }
  });
});
