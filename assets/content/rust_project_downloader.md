let semaphore = Arc::new(Semaphore::new(8));
let mut tasks = Vec::new();
for url in urls {
    let permit = semaphore.clone().acquire_owned().await?;
    tasks.push(tokio::spawn(async move {

    }));
}
for task in tasks { println!("{:?}", task.await?); }

### 测验回顾

1. Semaphore 许可被耗尽时，新下载任务会怎样？
   - 依据：在「实战：Rust 并发下载器」里，等待已有任务释放许可后再开始。acquireowned 在许可耗尽时会异步等待，直到某个任务完成并释放 permit。把“等待已有任务释放许可后再开始”代回「实战：Rust 并发下载器」里“Semaphore 许可被耗尽时”的例子核对，条件一旦改变，结论就要用Tokio、Semaphore、reqwest重新推导。
