int main(int argc, char** argv) {
  sqlite3* db = nullptr;

  sqlite3_exec(db, "CREATE TABLE IF NOT EXISTS tasks(id INTEGER PRIMARY KEY, title TEXT NOT NULL, done INTEGER DEFAULT 0);", nullptr, nullptr, nullptr);

  sqlite3_close(db);
}
