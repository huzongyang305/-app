func worker(ctx context.Context, jobs <-chan string, results chan<- Result, limiter *rate.Limiter) {
  for {
    select {
    case <-ctx.Done():
      return
    case url, ok := <-jobs:
      if !ok { return }

      results <- fetch(url)
    }
  }
}
```
