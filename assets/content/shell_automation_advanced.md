```bash
#!/usr/bin/env bash
set -euo pipefail

log() { printf '[%s] %s\n' "$1" "$2"; }

retry() {
  local attempts=0
  until "$@"; do
    attempts=$((attempts + 1))
    if (( attempts >= 3 )); then
      return 1
    fi
  done
}

retry true
log INFO "job done"
```
