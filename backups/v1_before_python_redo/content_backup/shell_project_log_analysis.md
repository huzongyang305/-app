#!/usr/bin/env bash
set -euo pipefail
LOG=${1:-/var/log/nginx/access.log}
Total=$(wc -l < "$LOG")
Errors=$(awk '$9 >= 500 {count++} END {print count+0}' "$LOG")

printf 'total=%s errors=%s error_rate=%s\n' "$Total" "$Errors" "$Rate"
