```sql
WITH RECURSIVE edges(src, dst) AS (
  VALUES ('a', 'b'), ('b', 'c'), ('c', 'd')
), walk(node, depth) AS (
  SELECT 'a', 0
  UNION ALL
  SELECT edges.dst, walk.depth + 1
  FROM edges
  JOIN walk ON edges.src = walk.node
)
SELECT node, depth FROM walk ORDER BY depth;
```
